package com.finnova.bank.audit;
import java.time.*;
import java.util.*;
import com.finnova.bank.auth.domain.Role;
import com.finnova.bank.banking.*;
import com.finnova.bank.banking.BankingDtos.PageView;
import com.finnova.bank.common.ApiException;
import com.finnova.bank.reports.DateWindow;
import jakarta.persistence.criteria.Predicate;
import org.springframework.http.HttpStatus;
import org.springframework.security.access.prepost.PreAuthorize;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
@Service
public class AuditQueryService {
    public record View(long id,String actorType,Long actorUserId,String actorRole,String eventType,String outcome,String resourceType,String resourceId,Map<String,Object> details,Instant occurredAt){}
    private final AuditRepository records;private final BankingAccess access;private final Clock clock;
    public AuditQueryService(AuditRepository records,BankingAccess access,Clock clock){this.records=records;this.access=access;this.clock=clock;}
    @Transactional(readOnly=true)@PreAuthorize("hasAnyRole('LOAN_OFFICER','ADMIN') and #actor==authentication.principal.id()")
    public PageView<View> search(long actor,Long actorId,String event,String outcome,String q,LocalDate from,LocalDate to,int page,int size){
        Role role=access.actor(actor);if(role==Role.LOAN_OFFICER&&actorId!=null&&actorId!=actor)throw new ApiException(HttpStatus.FORBIDDEN,"Officers can search only their own audit actions.");
        Long scoped=role==Role.LOAN_OFFICER?actor:actorId;String search=ListQuery.search(q);DateWindow range=DateWindow.of(from,to,clock);
        if(event!=null&&!event.matches("[A-Z0-9_]{1,64}"))throw new ApiException(HttpStatus.BAD_REQUEST,"Invalid event filter.");
        if(outcome!=null&&!Set.of("SUCCESS","FAILURE","DENIED").contains(outcome))throw new ApiException(HttpStatus.BAD_REQUEST,"Invalid outcome filter.");
        var query=ListQuery.of(page,size,"occurredAt","DESC",Set.of("occurredAt"));
        var result=records.findAll((r,c,b)->{List<Predicate> p=new ArrayList<>();p.add(b.greaterThanOrEqualTo(r.get("occurredAt"),range.start()));p.add(b.lessThan(r.get("occurredAt"),range.end()));
            if(scoped!=null)p.add(b.equal(r.get("actorUserId"),scoped));if(event!=null)p.add(b.equal(r.get("eventType"),event));if(outcome!=null)p.add(b.equal(r.get("outcome"),outcome));
            if(search!=null)p.add(b.or(b.like(b.lower(r.get("eventType")),ListQuery.like(search),'\\'),b.like(b.lower(r.get("resourceType")),ListQuery.like(search),'\\'),b.like(b.lower(r.get("resourceId")),ListQuery.like(search),'\\')));return b.and(p.toArray(Predicate[]::new));},query.pageable());
        return new PageView<>(result.getContent().stream().map(a->new View(a.getId(),a.getActorType(),a.getActorUserId(),a.getActorRole(),a.getEventType(),a.getOutcome(),a.getResourceType(),a.getResourceId(),a.getDetails(),a.getOccurredAt())).toList(),page,size,result.getTotalElements(),result.getTotalPages(),"occurredAt","DESC");
    }
}

    @Column(name = "occurred_at", nullable = false, updatable = false) private Instant occurredAt;
    @Column(name = "created_at", nullable = false, updatable = false) private Instant createdAt;

package com.finnova.bank.common;

import java.time.Clock;
import java.time.Instant;
import java.util.Comparator;
import java.util.List;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.dao.CannotAcquireLockException;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.dao.OptimisticLockingFailureException;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpStatus;
import org.springframework.http.HttpStatusCode;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.core.AuthenticationException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.context.request.ServletWebRequest;
import org.springframework.web.context.request.WebRequest;
import org.springframework.web.servlet.mvc.method.annotation.ResponseEntityExceptionHandler;

@RestControllerAdvice
public class GlobalExceptionHandler extends ResponseEntityExceptionHandler {
    private static final Logger LOG = LoggerFactory.getLogger(GlobalExceptionHandler.class);
    private final Clock clock;
    public GlobalExceptionHandler(Clock clock) { this.clock = clock; }

    @ExceptionHandler(ApiException.class)
    ResponseEntity<Object> business(ApiException ex, WebRequest request) {
        return error(ex.status(), ex.getMessage(), List.of(), new HttpHeaders(), request);
    }
    @ExceptionHandler(AccessDeniedException.class)
    ResponseEntity<Object> forbidden(AccessDeniedException ex, WebRequest request) {
        return error(HttpStatus.FORBIDDEN, "You do not have permission for this action.", List.of(), new HttpHeaders(), request);
    }
    @ExceptionHandler(AuthenticationException.class)
    ResponseEntity<Object> unauthenticated(AuthenticationException ex, WebRequest request) {
        return error(HttpStatus.UNAUTHORIZED, "Authentication is required or the session is no longer valid.", List.of(), new HttpHeaders(), request);
    }
    @ExceptionHandler(DataIntegrityViolationException.class)
    ResponseEntity<Object> integrity(DataIntegrityViolationException ex, WebRequest request) {
        return error(HttpStatus.CONFLICT, "The request conflicts with an existing record or an integrity rule.", List.of(), new HttpHeaders(), request);
    }
    @ExceptionHandler({CannotAcquireLockException.class, OptimisticLockingFailureException.class})
    ResponseEntity<Object> concurrentUpdate(Exception ex, WebRequest request) {
        return error(HttpStatus.CONFLICT, "The record changed or is busy. Retry the action.", List.of(), new HttpHeaders(), request);
    }
    @Override
    protected ResponseEntity<Object> handleMethodArgumentNotValid(MethodArgumentNotValidException ex,
            HttpHeaders headers, HttpStatusCode status, WebRequest request) {
        List<ApiError.FieldError> fields = ex.getBindingResult().getFieldErrors().stream()
                .map(field -> new ApiError.FieldError(field.getField(), field.getDefaultMessage()))
                .sorted(Comparator.comparing(ApiError.FieldError::field)).toList();
        return error(HttpStatus.BAD_REQUEST, "Request validation failed.", fields, headers, request);
    }
    @Override
    protected ResponseEntity<Object> handleExceptionInternal(Exception ex, Object body, HttpHeaders headers,
            HttpStatusCode status, WebRequest request) {
        String message = switch (status.value()) {
            case 400 -> "Malformed request or unsupported request fields.";
            case 404 -> "The requested resource was not found.";
            case 405 -> "This HTTP method is not supported for this resource.";
            case 415 -> "Use application/json for this request.";
            default -> "The request could not be processed.";
        };
        return error(HttpStatus.valueOf(status.value()), message, List.of(), headers, request);
    }
    @ExceptionHandler(Exception.class)
    ResponseEntity<Object> unexpected(Exception ex, WebRequest request) {
        // Never log request bodies, SQL, tokens, credentials, or exception messages.
        LOG.error("Unhandled API failure type={}", ex.getClass().getSimpleName());
        return error(HttpStatus.INTERNAL_SERVER_ERROR, "An unexpected error occurred. Please try again later.",
                List.of(), new HttpHeaders(), request);
    }
    private ResponseEntity<Object> error(HttpStatus status, String message, List<ApiError.FieldError> fields,
            HttpHeaders headers, WebRequest request) {
        String path = ((ServletWebRequest) request).getRequest().getRequestURI();
        ApiError error = new ApiError(Instant.now(clock), status.value(), status.getReasonPhrase(), message, path, fields);
        HttpHeaders safeHeaders = new HttpHeaders();
        safeHeaders.putAll(headers);
        safeHeaders.setCacheControl("no-store");
        return new ResponseEntity<>(error, safeHeaders, status);
    }
}
